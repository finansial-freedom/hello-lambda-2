name: CI

on:
  pull_request:
    branches: [dev, main]
  workflow_dispatch:

jobs:
  ci:
    runs-on: ubuntu-latest
    env:
      AWS_REGION: us-west-2
    steps:
      - name: Checkout code
        uses: actions/checkout@v99

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          version: '3.12'

      - name: Install linting tools
        run: pip install flake8

      - name: Python lint - hello-lambda
        run: flake8 app/app.py --max-line-length=120 --ignore=E501,W503

      - name: Build Docker image - hello-lambda
        run: docker build -t devops/hello-lambda:test .

      - name: Test Lambda handler imports - hello-lambda
        run: |
          docker run --rm devops/hello-lambda:test \
            python -c "import app; print('hello-lambda handler OK')"
