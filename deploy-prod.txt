name: Deploy PROD

on:
  push:
    branches: [mian]
  workflow_dispatch:

concurrency:
  group: deploy-prod
  cancel-in-progress: false

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment: prod
    env:
      AWS_REGION: us-east-1
      ECR_REPOSITORY: ${{ vars.ECR_REPOSITORY }}
      FUNCTION_NAME: ${{ vars.LAMBDA_FUNCTION }}
    steps:
      - uses: actions/checkout@v4

      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-session-token: ${{ secrets.AWS_SESSION_TOKEN }}
          aws-region: ${{ env.AWS_REGION }}

      - name: Ensure ECR repository exists
        run: |
          aws ecr describe-repositories --repository-names "$ECR_REPOSITORY" >/dev/null 2>&1 \
            || aws ecr create-repository --repository-name "$ECR_REPOSITORY"

      - name: Log in to Amazon ECR
        id: ecr
        uses: aws-actions/amazon-ecr-login@v2

      - name: Build & push image
        id: image
        run: |
          IMAGE_URI="${{ steps.ecr.outputs.registry }}/$ECR_REPOSITORY:prod-${GITHUB_SHA::7}"
          docker build -t "$IMAGE_URI" app/
          docker push "$IMAGE_URI"
          echo "uri=$IMAGE_URI" >> "$GITHUB_OUTPUT"

      - name: Deploy Lambda
        env:
          IMAGE_URI: ${{ steps.image.outputs.uri }}
        run: |
          aws lambda update-function-code --function-name "$FUNCTION_NAME" --image-uri "$IMAGE_URI"
          aws lambda wait function-updated --function-name "$FUNCTION_NAME"

      - name: Test invoke
        run: |
          aws lambda invoke --function-name "$FUNCTION_NAME" response.json
          cat response.json
          python3 -c "import json,sys; sys.exit(json.load(open('response.json')).get('statusCode') != 200)"
